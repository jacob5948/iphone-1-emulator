# The virtual Wi-Fi network's NAT (--net user) and the browser relay need
# libslirp. Both are optional: without it the harness still builds and the
# guest's Wi-Fi simply has no far end.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(SLIRP QUIET IMPORTED_TARGET slirp)
endif()
if(SLIRP_FOUND)
    add_library(netslirp STATIC net/net_slirp.c)
    target_link_libraries(netslirp PUBLIC PkgConfig::SLIRP)
    target_include_directories(netslirp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(iphone1 netslirp)
    target_compile_definitions(iphone1 PRIVATE IPHONE1_HAVE_SLIRP=1)
    add_executable(netrelay net/netrelay.c)
    target_link_libraries(netrelay netslirp)
endif()
